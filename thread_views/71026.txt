courtesy of chatgpt, might be a good starting point?

```
#include <stdlib.h> // For strtol
#include <stdbool.h> // For bool type

static int hawkbit_time2sec(const char *s) {
    long hours, minutes, seconds;
    char *next;

    // Validate and convert hours
    hours = strtol(s, &next, 10);
    if (*next != ':' || hours < 0) return -1;

    // Validate and convert minutes
    minutes = strtol(next + 1, &next, 10);
    if (*next != ':' || minutes < 0 || minutes >= 60) return -1;

    // Validate and convert seconds
    seconds = strtol(next + 1, &next, 10);
    if (*next != '\0' || seconds < 0 || seconds >= 60) return -1;

    return (hours * 3600) + (minutes * 60) + seconds;
}
```
