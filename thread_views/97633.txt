*--- josuah (2025-10-24 17:20:13+00:00)
If you wanted to simplify the driver, it's possible to use that:

```suggestion
	dqs_d_t = dqs_d_t > 15 ? 15 : dqs_d_t;
	dqs_d_t = (uint16_t)(0xFFFFFFFF0000ULL >> (dqs_d_t + 1));
```

To be sure it works, I've done this `test.c` and ran it with `cc -Wall -Wextra test.c && ./a.out`:
```c
#include <stdint.h>
#include <stdio.h>

#define ARRAY_SIZE(x) (sizeof(x) / sizeof(*(x)))

static const uint16_t dqs_delay_trims[16] = {
	0x8000,
	0xC000,
	0xE000,
	0xF000,
	0xF800,
	0xFC00,
	0xFE00,
	0xFF00,
	0xFF80,
	0xFFC0,
	0xFFE0,
	0xFFF0,
	0xFFF8,
	0xFFFC,
	0xFFFE,
	0xFFFF,
};

uint32_t test_prev(uint32_t dqs_d_t)
{
	dqs_d_t = dqs_d_t > ARRAY_SIZE(dqs_delay_trims) - 1 ?
		ARRAY_SIZE(dqs_delay_trims) - 1 : dqs_d_t;
	dqs_d_t = dqs_delay_trims[dqs_d_t];

	return dqs_d_t;
}

uint32_t test_next(uint32_t dqs_d_t)
{
	dqs_d_t = dqs_d_t > 15 ? 15 : dqs_d_t;
	dqs_d_t = (uint16_t)(0xFFFFFFFF0000ULL >> (dqs_d_t + 1));

	return dqs_d_t;
}

int main(void)
{
	for (uint32_t i = 0; i < 1000; i++) {
    		uint32_t prev = test_prev(i);
    		uint32_t next = test_next(i);

		if (prev != next) {
			printf("error at i=%u 0x%08x != 0x%08x\n", i, prev, next);
			return 1;
		}
	}

	printf("ok\n");

	return 0;
}
```
